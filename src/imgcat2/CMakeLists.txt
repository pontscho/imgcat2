# imgcat2 command line tool: imgcat2_cli (everything but main.c; also linked by the CLI tests) + imgcat2

set(_cli_src cli.c pipeline.c metadata.c base64.c ansi/ansi.c ansi/escape.c terminal/file.c)
if(WIN32)
	list(APPEND _cli_src terminal/terminal_windows.c)
else()
	list(APPEND _cli_src terminal/terminal_unix.c terminal/iterm2.c terminal/kitty.c)
endif()
add_library(imgcat2_cli STATIC ${_cli_src})
target_include_directories(imgcat2_cli PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(imgcat2_cli PUBLIC ${IMG2_LINK_TARGET})   # public <img2/...> API only; fat libimg2.a if built
if(TARGET img2_fat)
	add_dependencies(imgcat2_cli img2_fat)
endif()
target_compile_definitions(imgcat2_cli PRIVATE IMGCAT2_VERSION_STRING="${PROJECT_VERSION}")
img2_apply_project_settings(imgcat2_cli)

if(IMG2_BUILD_CLI)
	# [F30] Linux: static PIE (ASLR for the main image) when the toolchain supports it, else plain -static.
	# Needs PIC vendored archives (IMG2_POSITION_INDEPENDENT_CODE). Option and probe live here (CLI only), so
	# an add_subdirectory() consumer's cache never sees them.
	set(_imgcat2_pie_arg "")
	if(IMG2_SANITIZE)
		message(STATUS "imgcat2: sanitizer build, dynamic PIE link (no -static / -static-pie)")
	elseif(UNIX AND NOT APPLE)
		option(IMG2_STATIC_PIE "Link imgcat2 as a static PIE (-static-pie) on Linux when supported" ON)
		if(IMG2_STATIC_PIE AND NOT IMG2_POSITION_INDEPENDENT_CODE)
			message(STATUS "imgcat2: IMG2_POSITION_INDEPENDENT_CODE is OFF, linking with plain -static (no static PIE)")
		elseif(IMG2_STATIC_PIE)
			include(CheckCSourceCompiles)
			include(CMakePushCheckState)
			cmake_push_check_state(RESET)
			set(CMAKE_REQUIRED_FLAGS -fPIE)
			set(CMAKE_REQUIRED_LINK_OPTIONS -static-pie)
			check_c_source_compiles("#include <stdio.h>\nint main(void){puts(\"img2\");return 0;}" IMG2_HAVE_STATIC_PIE)
			cmake_pop_check_state()
			if(IMG2_HAVE_STATIC_PIE)
				set(_imgcat2_pie_arg STATIC_PIE)
				set_target_properties(imgcat2_cli PROPERTIES POSITION_INDEPENDENT_CODE ON)
				message(STATUS "imgcat2: linking as static PIE (-static-pie)")
			else()
				message(STATUS "imgcat2: toolchain does not support -static-pie, linking with plain -static")
			endif()
		endif()
	endif()

	add_executable(imgcat2 main.c)
	target_link_libraries(imgcat2 PRIVATE imgcat2_cli)
	set_target_properties(imgcat2 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")   # keeps build/imgcat2
	img2_apply_project_settings(imgcat2)
	img2_static_executable(imgcat2 ${_imgcat2_pie_arg})
	# [SL2] Full RELRO (exe only; works for the static PIE and the plain static link)
	target_link_options(imgcat2 PRIVATE $<$<PLATFORM_ID:Linux>:-Wl,-z,relro,-z,now>)
	# install rule: cmake/Img2Install.cmake (COMPONENT imgcat2)
endif()
